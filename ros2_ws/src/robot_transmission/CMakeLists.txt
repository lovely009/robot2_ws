cmake_minimum_required(VERSION 3.16)
project(robot_transmission)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
find_package(transmission_interface REQUIRED)
find_package(hardware_interface REQUIRED)
find_package(pluginlib REQUIRED)
find_package(yaml_cpp_vendor REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter Development)
find_package(pybind11 REQUIRED)
add_library(${PROJECT_NAME} SHARED src/model.cpp src/spatial_linkage.cpp)
pluginlib_export_plugin_description_file(transmission_interface transmission_plugins.xml)
target_compile_features(${PROJECT_NAME} PUBLIC cxx_std_17)
target_compile_options(${PROJECT_NAME} PRIVATE -Wall -Wextra -Wpedantic)
target_include_directories(${PROJECT_NAME} PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
ament_target_dependencies(${PROJECT_NAME} transmission_interface hardware_interface pluginlib)
target_link_libraries(${PROJECT_NAME} yaml-cpp)
add_library(robot_transmission_plugins SHARED src/spatial_linkage_loader.cpp)
ament_target_dependencies(robot_transmission_plugins pluginlib)
target_link_libraries(robot_transmission_plugins ${PROJECT_NAME})
# 插件不导出给核心库调用方链接，必须由 pluginlib 独立加载。
install(TARGETS robot_transmission_plugins LIBRARY DESTINATION lib)
ament_python_install_package(${PROJECT_NAME})
ament_get_python_install_dir(PYTHON_INSTALL_DIR)
pybind11_add_module(_core src/python.cpp)
target_link_libraries(_core PRIVATE ${PROJECT_NAME})
install(TARGETS _core LIBRARY DESTINATION ${PYTHON_INSTALL_DIR}/${PROJECT_NAME})
install(TARGETS ${PROJECT_NAME} EXPORT export_${PROJECT_NAME} ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
install(DIRECTORY include/ DESTINATION include)
ament_export_targets(export_${PROJECT_NAME} HAS_LIBRARY_TARGET)
ament_export_dependencies(transmission_interface hardware_interface pluginlib yaml_cpp_vendor)
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_pytest_test(test_python test/test_python.py)
  ament_add_gtest(test_model test/test_model.cpp)
  target_link_libraries(test_model ${PROJECT_NAME})
  ament_add_gtest(test_spatial_linkage test/test_spatial_linkage.cpp)
  target_link_libraries(test_spatial_linkage ${PROJECT_NAME})
endif()
ament_package()
